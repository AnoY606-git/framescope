add_library(framescope_options INTERFACE)
target_compile_definitions(framescope_options INTERFACE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
  _WIN32_WINNT=0x0A00 FRAMESCOPE_VERSION="${FRAMESCOPE_VERSION}" $<$<CONFIG:Debug>:_DEBUG>)
if(MSVC)
  target_compile_options(framescope_options INTERFACE /W4 /permissive- /utf-8 /EHsc)
else()
  target_compile_options(framescope_options INTERFACE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
  target_link_options(framescope_options INTERFACE -static-libstdc++ -static-libgcc)
endif()

add_library(framescope_common STATIC
  native/common/bridge.cpp native/common/statistics.cpp native/common/hud.cpp native/common/fonts.cpp)
target_include_directories(framescope_common PUBLIC native/include)
target_link_libraries(framescope_common PUBLIC framescope_options imgui)

function(framescope_version_resource target suffix file_type description)
  set(FRAMESCOPE_RESOURCE_FILENAME "${target}.${suffix}")
  set(FRAMESCOPE_RESOURCE_TYPE "${file_type}")
  set(FRAMESCOPE_RESOURCE_DESCRIPTION "${description}")
  set(resource "${CMAKE_CURRENT_BINARY_DIR}/resources/${target}.rc")
  configure_file(native/common/version.rc.in "${resource}" @ONLY)
  target_sources(${target} PRIVATE "${resource}")
endfunction()

  framescope_version_resource(FrameScope.Hud dll VFT_DLL "FrameScope DirectX HUD")
